# Задание 2

Команды
dhall-to-json < task2.dhall > task2.json

Код dhall файла
```dhall
let List/generate = https://prelude.dhall-lang.org/v23.1.0/List/generate

let makeGroup = \(n : Natural) -> "ИКБО-${Natural/show n}-20"

let groups = List/generate 24 Text (\(n : Natural) -> makeGroup (n + 1))

in  { groups = groups
    , students =
      [ { age = 19, group = "ИКБО-4-20", name = "Иванов И.И." }
```
